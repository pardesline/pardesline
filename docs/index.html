import pardesline as pl

ds = pl.load("msd_task02_heart")                 # 30 cardiac MRIs, nothing downloaded yet
s = ds.filter(split="Tr")[0]                     # roles: mri, mask - licence attached
atrium = s.mask.to_mesh(label=1)                 # exact left-atrium surface, LPS millimetres
atrium.decimate(target_reduction=0.5).save("atrium.stl")

ct = pl.load("tcia://Pancreas-CT")[0].ct         # a DICOM series from The Cancer Imaging Archive
moved = ct.align(pl.load("other.nii.gz"))        # rigid registration, transform in moved.transform
ct.to_sitk(), atrium.to_open3d(), atrium.to_pyvista()   # the libraries you already use
